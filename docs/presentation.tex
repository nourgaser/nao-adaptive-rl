\section{Learning Paradigms}

\begin{frame}{Three ways a machine can learn}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \textbf{Supervised}
      \begin{itemize}\footnotesize
        \item Learn input $\to$ correct label from examples.
        \item Needs large \emph{pre-labeled} data; trained \emph{offline}.
        \item E.g.\ emotion classifier from labeled images.
      \end{itemize}
    \end{column}
    \begin{column}{0.33\textwidth}
      \textbf{Unsupervised}
      \begin{itemize}\footnotesize
        \item Find structure in \emph{unlabeled} data.
        \item Clustering, dimensionality reduction.
        \item E.g.\ group users by interaction style.
      \end{itemize}
    \end{column}
    \begin{column}{0.33\textwidth}
      \textbf{\alert{Reinforcement}}
      \begin{itemize}\footnotesize
        \item Learn by \emph{trial and error} from a \emph{reward}.
        \item No labels; adapts \emph{online}.
        \item E.g.\ adjust distance if the person leans in / backs away.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{alertblock}{Why not just supervised learning?}
    The ``correct'' social behavior is person- and context-specific, unknown in advance, and shifts \emph{during} interaction. Supervised learning needs a labeled answer up front and cannot adapt mid-interaction. RL learns the answer \emph{from} the interaction.
  \end{alertblock}
\end{frame}

\section{Reinforcement Learning in Depth}

\begin{frame}{The RL loop: agent, environment, reward}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \centering
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[
        node distance=1.4cm,
        box/.style={draw, rounded corners, minimum width=2.6cm, minimum height=1cm, align=center, thick},
        every node/.style={font=\footnotesize}]
        \node[box, fill=giublue!12] (agent) {\textbf{Agent}\\(NAO)};
        \node[box, fill=giuaccent!12, below=1.8cm of agent] (env) {\textbf{Environment}\\(person + world)};
        \draw[-{Stealth[length=3mm]}, thick]
          (agent.east) to[bend left=45] node[right, midway]{action $A_t$} (env.east);
        \draw[-{Stealth[length=3mm]}, thick]
          (env.west) to[bend left=45] node[left, midway, align=center]{state $S_{t+1}$\\reward $R_{t+1}$} (agent.west);
      \end{tikzpicture}%
      }
    \end{column}
    \begin{column}{0.50\textwidth}
      At each step $t$ the agent:
      \begin{enumerate}\footnotesize
        \item observes state $S_t$,
        \item takes action $A_t$ via policy $\pi(a\mid s)$,
        \item receives reward $R_{t+1}$,
        \item lands in state $S_{t+1}$.
      \end{enumerate}
      \medskip
      \textbf{Goal:} find $\pi$ maximizing expected \emph{discounted return}
      \[
        G_t=\textstyle\sum_{k\ge 0}\gamma^{k}R_{t+k+1},\quad \gamma\in[0,1).
      \]
      \textbf{Value} $Q(s,a)$ = expected return from taking $a$ in $s$.
    \end{column}
  \end{columns}
  \vfill
  {\footnotesize\raggedright Formalized as an MDP $\langle S,A,P,R,\gamma\rangle$: the next state depends only on current $(s,a)$ --- the \emph{Markov property}. \refnum{Sutton2018}\par}
\end{frame}

\begin{frame}{Q-learning: the workhorse for \emph{simple} adaptive behavior}
  \textbf{Idea:} learn the value of each (state, action) pair directly --- no model of the world needed.
  \vspace{0.4em}
  \begin{block}{Q-learning update (Watkins \& Dayan, 1992)}
    \[
      Q(s,a)\;\leftarrow\;Q(s,a)\;+\;\alpha\Big[\,\underbrace{r+\gamma\max_{a'}Q(s',a')}_{\text{target}}\;-\;Q(s,a)\,\Big]
    \]
    \footnotesize $\alpha$: learning rate \quad $\gamma$: discount \quad $r$: observed reward
  \end{block}
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Exploration vs.\ exploitation}
      \begin{itemize}\footnotesize
        \item Must try new actions \emph{and} use known-good ones.
        \item \alert{$\varepsilon$-greedy}: random action w.p.\ $\varepsilon$, else $\arg\max_a Q(s,a)$; decay $\varepsilon$ over time.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Tabular vs.\ approximation}
      \begin{itemize}\footnotesize
        \item \emph{Tabular}: one value per $(s,a)$ --- perfect for small, discrete spaces (our case).
        \item Large/continuous $\to$ function approximation (Deep Q-Networks).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Policy gradients \& learning from \emph{human} reward}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Policy gradient (REINFORCE)}
      \begin{itemize}\footnotesize
        \item Directly tune a parameterized policy $\pi_\theta$ toward higher reward.
        \item Natural fit for \emph{continuous} action parameters: distance, speed, timing.
        \item Used by Mitsunaga et al.\ to adapt proxemics, gaze \& timing on a social robot. \refnum{Mitsunaga2005}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Interactive RL / human feedback}
      \begin{itemize}\footnotesize
        \item The \emph{human} supplies the reward signal.
        \item \alert{TAMER}: non-experts shape an agent via approval / disapproval. \refnum{Knox2009}
        \item Cuts the number of trials needed --- crucial on hardware.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{alertblock}{The hardware reality (a recurring theme)}
    Every real trial costs \emph{time}, \emph{battery}, and \emph{joint wear}. So \textbf{sample efficiency} matters, and \textbf{sim-to-real transfer} (the ``reality gap'') is a genuine risk: policies that work in simulation can fail on the physical robot.
  \end{alertblock}
\end{frame}

\begin{frame}{Learning from Demonstration (LfD) --- a complement to RL}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Teach by showing}, not by coding or rewarding. \refnum{Argall2009}
      \begin{itemize}
        \item Intuitive for non-experts --- people teach by demonstrating.
        \item \alert{Kinesthetic teaching}: set joint stiffness to zero, physically move NAO's limb, record the joint trajectory, replay it.
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{block}{Why pair LfD with RL?}
        A demonstration gives a good \emph{starting policy}; RL then \emph{refines} it through interaction --- fixing RL's slow cold-start and poor sample efficiency.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Personalization \& Adaptation}

\begin{frame}{Personalization in long-term HRI}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{Why adapt at all?}
      \begin{itemize}\footnotesize
        \item Short-term studies hide the hard part: sustaining engagement over many sessions. \refnum{Irfan2019}
        \item Adaptation counters the \alert{novelty effect} --- interest fades once a robot stops being new.
        \item Built on \emph{user modeling} + \emph{incremental / lifelong learning}. \refnum{Irfan2021}
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{Static vs.\ dynamic}
      \begin{itemize}\footnotesize
        \item \emph{Static}: fix a profile once (a questionnaire).
        \item \alert{Dynamic}: adjust in real time to live social signals --- our RL approach.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{alertblock}{The ethical edge: entertainment vs.\ manipulation \refnum{Pollmann2023}}
    The same personalization that improves user experience can be used to \emph{steer} a person against their own autonomy. Adaptive systems need explicit ethical guardrails --- a point we will state up front, not bolt on.
  \end{alertblock}
\end{frame}

\begin{frame}{Paper 1 --- Gonzalez-Santocildes et al.\ (2024): the problem}
  \footnotesize \emph{``Adaptive Robot Behavior Based on Human Comfort Using Reinforcement Learning''}, IEEE Access. \refnum{Gonzalez2024}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{The problem}
      \begin{itemize}\footnotesize
        \item Most RL-for-robotics ignores the \emph{human}: optimizes task success, not human comfort.
        \item Prior comfort work is largely \emph{theoretical} and \alert{not replicable}.
      \end{itemize}
      \textbf{Their contribution}
      \begin{itemize}\footnotesize
        \item An open-source, parametric \alert{Unity} environment the community can reuse.
        \item A \emph{comfort reward function} whose strictness is set by one parameter.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{The setup (important caveats)}
      \begin{itemize}\footnotesize
        \item Robot: \alert{UR3e} --- an \emph{industrial arm}, not a social/humanoid robot.
        \item Task: pick-and-place a coloured ball while a (simulated) human moves nearby.
        \item Comfort here = \emph{physical} (distance, speed), \emph{not social} (warmth, rapport).
        \item Algorithm: \alert{SAC} (deep RL for continuous control).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 1 --- Methodology: comfort inside the reward}
  \footnotesize
  Standard task reward, \emph{minus} a comfort penalty that the user can switch on and weight:
  \[
    R = k_1 F_{\text{task1}} + k_2 F_{\text{task2}} + k_3 F_{\text{task3}} \;-\; F_{\text{comfort}}(C_l, N_c)
  \]
  \[
    F_{\text{comfort}}(C_l,N_c) = \alpha\, F_{\text{conf1}}(C_l,N_c)\;+\;\beta\, F_{\text{conf2}}(C_l,N_c)
  \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\footnotesize
        \item $F_{\text{conf1}}$: penalty on \alert{distance} (tanh threshold) --- too close hurts reward.
        \item $F_{\text{conf2}}$: penalty on \alert{joint speed} per step --- too fast hurts reward.
        \item $\alpha,\beta$: weight each part; set both $=0$ and it is plain task RL.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Why ``comfort \emph{levels}'' ($C_l$ of $N_c$)?}
      \begin{itemize}\footnotesize
        \item Comfort is \alert{discretized} (10 levels), each trained separately --- easier than one policy over a continuum.
        \item The level sets \emph{where} the distance/speed penalty threshold sits.
        \item Static per run = \emph{controlled variation}: vary one thing, attribute the change cleanly.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 1 --- Results: one parameter, measurably different behavior}
  \begin{center}

\begin{frame}{Paper 1 --- Critique \& what it means for us}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Strengths}
      \begin{itemize}\footnotesize
        \item Clean \emph{controlled} result: behavior provably tracks the comfort parameter.
        \item Open, replicable environment.
      \end{itemize}
      \textbf{Gaps (their own + ours)}
      \begin{itemize}\footnotesize
        \item \alert{Comfort is set manually}; measuring it \emph{live} from a person is unsolved (they suggest HR/EEG/a button).
        \item Only \emph{move-pick-place} tasks --- does it generalize?
        \item \emph{Physical/industrial} comfort, not \emph{social}; not a humanoid.
        \item Validated only in \emph{simulation} --- real-robot transfer is future work.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{How this feeds our project}
      \begin{itemize}\footnotesize
        \item We take their proven idea --- \emph{comfort in the reward} --- and supply the missing piece: a \alert{live} comfort signal from NAO's \emph{sonar + tactile} sensors.
        \item We move from \emph{physical} to \alert{social} comfort (greeting distance), bridging to Mitsunaga.
        \item Discrete states $\Rightarrow$ \emph{tabular Q-learning}, not SAC: lighter, laptop-trainable, faster.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 2 --- Zedda et al.\ (2023): the problem}
  \footnotesize \emph{``An Adaptive Behaviour-Based Strategy for SARs interacting with Older Adults with MCI during a Serious Game Scenario''}, SARs workshop @ CHI 2023. \refnum{Zedda2023}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{The problem}
      \begin{itemize}\footnotesize
        \item Cognitive training for older adults with \alert{MCI} is repetitive $\Rightarrow$ boredom $\Rightarrow$ disengagement $\Rightarrow$ dropout.
        \item A socially assistive robot (SAR) could keep them engaged --- \emph{if} it adapts its behavior to the user's state.
      \end{itemize}
      \textbf{Their contribution}
      \begin{itemize}\footnotesize
        \item A \alert{Q-learning} strategy that picks robot \emph{verbal + non-verbal} actions to keep engagement high.
        \item Two selectable robot \emph{personalities} (extravert / introvert).
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{The setup (caveats)}
      \begin{itemize}\footnotesize
        \item Robot: \alert{Pepper} --- NAO's sibling (same maker, same \texttt{NAOqi} SDK) $\Rightarrow$ transfers to NAO.
        \item Task: a ``cooking'' serious game (8 questions).
        \item \alert{Simulation only}: two user models (healthy + MCI); real-user study is future work.
        \item This is the \emph{workshop} precursor; a fuller 2025 journal version exists.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 2 --- Methodology: the engagement MDP}
  \footnotesize Tabular Q-learning, $Q(s,a)$ table, $\varepsilon$-greedy; model-free, off-policy.
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \textbf{State $S$}
      \begin{itemize}\footnotesize
        \item Gaze direction (robot / tablet / away)
        \item Smiling (none / smile / broad)
        \item Answer (right / wrong)
        \item $\Rightarrow$ engagement level
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Action $A$ (3)}
      \begin{itemize}\footnotesize
        \item $a_0$: enthusiastic / dynamic
        \item $a_1$: neutral
        \item $a_2$: stimulating (re-engage)
      \end{itemize}
      {\scriptsize each filtered through personality: volume, rate, pitch, gesture size}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Reward $R$}
      \begin{itemize}\footnotesize
        \item $+$ keep engagement medium/high
        \item $-$ let it fall to ``low''
        \item $-0.05$ per step (settle fast)
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{alertblock}{A sharp detail worth noting}
    \footnotesize They set discount $\gamma=0.05$ --- \emph{very} myopic: the agent almost only values the \emph{immediate} engagement response. Reasonable for a per-question reward, but it forgoes long-horizon credit assignment (it behaves almost like a contextual bandit). Learning rate $\alpha=0.8$ is high.
  \end{alertblock}
\end{frame}

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 3 --- Tozadore et al.\ (2018): the problem}
  \footnotesize \emph{``Towards adaptation and personalization in task based on Human-Robot Interaction''}, LARS/SBR 2018. \refnum{Tozadore2018}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{The problem}
      \begin{itemize}\footnotesize
        \item Social robots lose \alert{novelty} $\Rightarrow$ motivation \& attention drop.
        \item Hypothesis: personalized conversation + content adaptation holds interest longer.
      \end{itemize}
      \textbf{Their contribution}
      \begin{itemize}\footnotesize
        \item A \alert{dynamic} user-adaptation system (vs.\ the usual \emph{static} user models).
        \item Reads body + verbal signals to adjust behavior; stores user info to personalize dialogue (rapport).
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{The setup (key facts)}
      \begin{itemize}\footnotesize
        \item Robot: \alert{NAO} --- \emph{our exact platform}.
        \item Task: a storytelling activity; \textbf{10 real participants}.
        \item \alert{Decision-making is RULE-BASED}, not RL --- ML is used only for \emph{perception}.
        \item This is the ``before-RL'' approach: hand-tuned weighted rules.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 3 --- Methodology: hand-tuned adaptation + ML perception}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Decision logic (rule-based)}
      \footnotesize
      Three indicator groups $\rightarrow$ parameters:
      \begin{itemize}\footnotesize
        \item Attention $\alpha$: face gaze, posture
        \item Communication $\beta$: words, emotion
        \item Task $\gamma$: success rate, answer time
      \end{itemize}
      \[ F_{\text{Adp}}(t)=w_\alpha\alpha + w_\beta\beta + w_\gamma\gamma \]
      {\scriptsize $\ge 0.66 \to {+}1$, $\le 0.33 \to {-}1$, else $0$; behavior $\Psi\!\in\!\{1..5\}$ (info $\leftrightarrow$ gesture), starts at 3.}
      {\scriptsize Weights set by hand: $w_\gamma{=}0.6$ (task), $w_\alpha{=}w_\beta{=}0.2$.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Perception (the ML part)}
      \begin{itemize}\footnotesize
        \item Face/gaze: \alert{Haar cascade} (attention deviations).
        \item Emotion: \alert{MobileNet}, Ekman 6+neutral; trained on FER+, RAF, QIDER.
        \item Answer check: \alert{Levenshtein} edit distance.
      \end{itemize}
      \textbf{User model (personalization)}
      \begin{itemize}\footnotesize
        \item Stores name, age, 8 preferences (sport, food, band\ldots).
        \item Pulls definitions from \alert{Wikipedia} to chat about them; gives NAO a ``personal life'' for rapport.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 3 --- Critique \& what it means for us}

        \item Genuine personalization (preferences, rapport) + honest negative findings.

\begin{frame}{Paper 4 --- Wang et al.\ (2025): the problem}
  \footnotesize \emph{``Personalization in HRI through Preference-based Action Representation Learning'' (PbARL)}, arXiv 2025. \refnum{Wang2025}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{The big shift: learn the reward}
      \begin{itemize}\footnotesize
        \item All prior papers \emph{hand-design} the reward (comfort, engagement, weights). Hard and brittle.
        \item \alert{Preference-based RL (PbRL)}: show the person two behaviors, ask ``which do you prefer?''; \emph{learn} the reward from those comparisons.
      \end{itemize}
      \textbf{The problem they fix}
      \begin{itemize}\footnotesize
        \item Existing PbRL trains a policy \alert{from scratch} per user $\Rightarrow$ wastes human feedback on the basic task.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{The setup (caveats)}
      \begin{itemize}\footnotesize
        \item Robot: \alert{Kinova Jaco} assistive arm (not humanoid/NAO).
        \item Assistive Gym sim + real user study (\textbf{N=8}).
        \item \alert{Frontier \& heavy}: deep RL, a cVAE, \textbf{3$\times$ RTX 4090 GPUs} --- the opposite end from our laptop tabular approach.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 4 --- Methodology: personalize without breaking the task}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{The core idea}
      \begin{itemize}\footnotesize
        \item Start from a \alert{frozen} pre-trained policy that already knows the basic task.
        \item Don't retrain it (that degrades the task). Instead, learn a \alert{latent action representation} (a conditional VAE) layered on top.
        \item Train it to \emph{maximize mutual information} between the task domain and the preference-aligned domain.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Why it works}
      \begin{itemize}\footnotesize
        \item Naive fine-tuning optimizes \emph{only} preference $\Rightarrow$ breaks the task.
        \item PbARL balances both: reconstruction loss \emph{preserves the task}, contrastive loss \emph{aligns preferences}.
      \end{itemize}
      \begin{block}{}
        \scriptsize Far more complex than tabular Q-learning --- we cite it as the \emph{state of the art}, not as something we replicate.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paper 4 --- Results}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\linewidth]{wang_userstudy.pdf}
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
      \textbf{Benchmark (Assistive Gym)}
      \begin{itemize}\footnotesize
        \item PbARL keeps task success (only $\downarrow$4--7) \emph{and} boosts preference reward ($\sim$+40).
        \item Naive fine-tune (PrefFT) personalizes but drops success $\downarrow$11--22; from-scratch PbRL \alert{collapses} ($\downarrow$48--53).
      \end{itemize}
      \textbf{Real study (N=8)}: PbARL best on \alert{both} satisfaction \& personalization ($p<.01$).
    \end{column}

\begin{frame}{Paper 4 --- Critique \& what it means for us}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Strengths}
      \begin{itemize}\footnotesize
        \item \alert{Learns} the reward from preferences --- no hand-design.
        \item Personalizes \emph{without} degrading the task; real IRB-approved user study.
      \end{itemize}
      \textbf{Gaps (their own + ours)}
      \begin{itemize}\footnotesize
        \item Assistive arm, not humanoid; \alert{heavy compute} (3 GPUs).
        \item Complex pipeline; preferences treated as fixed (lifelong = future work); $N=8$.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{How this feeds our project}
      \begin{itemize}\footnotesize
        \item Defines the \alert{frontier}: where reward design is going (learned from preferences).
        \item Justifies our scope by contrast: we start \emph{simple, interpretable, laptop-trainable} (tabular Q-learning + explicit tactile reward).
        \item Clear \alert{future work} for us: replace our hand-set reward with \emph{learned preferences}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Synthesis --- methodological gaps across the literature}
  \begin{itemize}
    \item \textbf{Reality gap:} fast convergence in simulation, but sensor noise and dynamics differ on hardware.
    \item \textbf{Sample efficiency:} real trials are slow and cause wear --- few-shot learning, human reward, and LfD warm-starts help.
    \item \textbf{Reward specification:} ``social reward'' is noisy and hard to define; bad rewards yield awkward behavior.
    \item \textbf{Static vs.\ dynamic preference:} treating preference as fixed fails long-term.
    \item \textbf{Evaluation:} over-reliance on subjective post-hoc surveys; need to \emph{combine} objective sensor metrics with subjective scales.
    \item \textbf{Generalization \& reproducibility:} per-user policies may not transfer; HRI lacks standardized metrics/protocols.
  \end{itemize}
  \begin{block}{}
    \footnotesize These gaps are our \textbf{Q\&A defense} --- for each we will hold a one-line position grounded in the literature.
  \end{block}
\end{frame}

\begin{frame}{Appendix 1 --- One Q-learning update, step by step}
  \footnotesize
  \textbf{Setup (Option A greeting MDP).} $\alpha=0.5$, $\gamma=0.9$. Current estimate $Q(s_{\text{far}},\,\texttt{closer})=0.0$.
  \begin{enumerate}
    \item \textbf{State} $s_{\text{far}}$: person in the \emph{social} distance bin.
    \item \textbf{Action} (via $\varepsilon$-greedy, exploiting): $a=\texttt{move closer}$.
    \item \textbf{Outcome:} person stays and gives a head touch $\Rightarrow$ reward $r=+1$.
    \item \textbf{Next state} $s'=s_{\text{personal}}$, where current best value is $\max_{a'}Q(s',a')=0.4$.
  \end{enumerate}
  \vspace{0.3em}
  \begin{block}{Apply the update}
    \[
      Q(s_{\text{far}},\texttt{closer}) \leftarrow 0.0 + 0.5\big[\,1 + 0.9(0.4) - 0.0\,\big]
      = 0.5\,(1.36) = \alert{0.68}
    \]
  \end{block}
  \footnotesize The estimate rises from $0.0$ to $0.68$: ``moving closer from far \emph{was} good here.'' Repeat over episodes and the table converges to a policy that approaches to each person's comfortable distance. \refnum{Sutton2018}
\end{frame}

\begin{frame}{Appendix 2 --- Supervised vs.\ RL: the same NAO task}
  \footnotesize
  \textbf{Task: choose how close NAO stands when greeting a new person.}
  \begin{table}
    \begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.1cm}>{\raggedright\arraybackslash}p{4.1cm}>{\raggedright\arraybackslash}p{4.7cm}@{}}
      \toprule
       & \textbf{Supervised learning} & \textbf{Reinforcement learning} \\
      \midrule
      Data needed & A large labeled set: (person features $\to$ ``correct'' distance), gathered in advance & None up front --- learns from live reward (touch, sonar, dwell) \\
      \addlinespace
      When it learns & Offline, before deployment & Online, \emph{during} interaction \\
      \addlinespace
      New / unseen person & Predicts from the fixed model; cannot adjust if wrong & Tries, observes feedback, corrects within the session \\
      \addlinespace
      Failure mode & Confidently wrong on anyone unlike the training data & Some early exploration cost before convergence \\
      \bottomrule
    \end{tabular}
  \end{table}
  \vspace{0.2em}
  \alert{Takeaway:} the ``right'' social distance is personal and unknown in advance --- exactly the setting where RL's online adaptation beats a pre-trained classifier.
\end{frame}

\begin{frame}{Appendix 3 --- Ethics: entertainment vs.\ manipulation}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{The risk \refnum{Pollmann2023}}
      \begin{itemize}\footnotesize
        \item Personalization that improves UX can also \emph{steer} behavior against a person's autonomy.
        \item A robot optimizing ``engagement'' could learn manipulative tactics if the reward is naive.
        \item Vulnerable users (children, elderly) raise the stakes.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Our safeguards}
      \begin{itemize}\footnotesize
        \item Reward ties to \emph{comfort}, not raw attention-maximization.
        \item Transparency: the person knows the robot is adapting.
        \item Informed \alert{consent} + right to stop, for every study participant.
        \item Data minimization: store only what the task needs.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{alertblock}{Question for the committee}
    What is the required ethics / consent procedure for running the user study with participants on campus?
  \end{alertblock}
\end{frame}

\begin{frame}{Appendix 4 --- The reality gap: will our sim results transfer?}
  \footnotesize
  We prototype in simulation \emph{first} --- so we must be honest about what that does and does not prove.
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Why a gap exists}
      \begin{itemize}
        \item Real sonar / tactile readings are \emph{noisy}; mocks are clean.
        \item Real humans behave unpredictably; scripted ``users'' do not.
        \item Joint dynamics, latency, and timing differ on hardware.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{How we mitigate}
      \begin{itemize}
        \item Inject noise into the mock sensors to harden the policy.
        \item Keep the state/action space small \& discrete (robust to noise).
        \item Treat sim as \emph{convergence proof}; reserve real trials for \emph{fine-tuning}.
        \item Report sim and real results \emph{separately} --- never conflate them.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{block}{}
    Honest framing: simulation shows the \emph{algorithm} works; only the physical NAO shows the \emph{interaction} works.
  \end{block}
\end{frame}
